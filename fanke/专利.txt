  border-radius: 8px;
  font-size: 13px;
  font-weight: 600;
  box-shadow: 0 10px 25px rgba(0, 0, 0, 0.2);
  opacity: 0;
  pointer-events: none;
  transition: all 0.25s ease;
  z-index: 999999;
}
#wxz-patent-app .wxz-toast.show {
  opacity: 1;
  transform: translateX(-50%) translateY(0);
}
</style>

<script>
/**
 * 王小治教授课题组专利展示系统 前端逻辑
 * 优先引用 https://wxz.group/patents_data.js
 */
(function() {
  var App = {
    data: [],
    stats: null,
    currentSort: 'default', // 'default' | 'date' | 'status'
    searchQuery: '',
    filters: {
      status: 'all',  // 'all' | 'granted' | 'filed'
      role: 'all',    // 'all' | 'first' | 'co'
      type: 'all',    // 'all' | 'inv' | 'utility'
      year: 'all'     // 'all' | '2026' ... | 'lte2016'
    },
    expandedAbstracts: {},

    init: function() {
      var self = this;
      self.bindGlobalEvents();
      self.loadData(function() {
        self.applyFilters();
      });
    },

    loadData: function(callback) {
      var self = this;
      if (window.WXZ_PATENTS_DATA && window.WXZ_PATENTS_STATS) {
        self.data = window.WXZ_PATENTS_DATA;
        self.stats = window.WXZ_PATENTS_STATS;
        if (callback) callback();
        return;
      }

      // 优先从 https://wxz.group/patents_data.js 加载
      var script = document.createElement('script');
      script.src = 'https://wxz.group/patents_data.js';
      script.onload = function() {
        if (window.WXZ_PATENTS_DATA) {
          self.data = window.WXZ_PATENTS_DATA;
          self.stats = window.WXZ_PATENTS_STATS;
          if (callback) callback();
        }
      };
      script.onerror = function() {
        // 本地环境或无网络时回退到 ./patents_data.js
        var fallbackScript = document.createElement('script');
        fallbackScript.src = './patents_data.js';
        fallbackScript.onload = function() {
          if (window.WXZ_PATENTS_DATA) {
            self.data = window.WXZ_PATENTS_DATA;
            self.stats = window.WXZ_PATENTS_STATS;
            if (callback) callback();
          }
        };
        fallbackScript.onerror = function() {
          var listEl = document.getElementById('wxz-pat-list');
          if (listEl) {
            listEl.innerHTML = '<div style="padding:40px;text-align:center;color:#ef4444;font-size:14px;">未能读取 patents_data.js 数据文件，请检查网络或确认文件已上传至 https://wxz.group/patents_data.js。</div>';
          }
        };
        document.head.appendChild(fallbackScript);
      };
      document.head.appendChild(script);
    },

    bindGlobalEvents: function() {
      var self = this;
      var root = document.getElementById('wxz-patent-app');
      if (!root) return;

      var searchInput = document.getElementById('wxz-search-input');
      var searchClear = document.getElementById('wxz-search-clear');
      if (searchInput) {
        searchInput.addEventListener('input', function(e) {
          self.searchQuery = e.target.value.trim().toLowerCase();
          if (searchClear) searchClear.style.display = self.searchQuery ? 'block' : 'none';
          self.applyFilters();
        });
      }
      if (searchClear) {
        searchClear.addEventListener('click', function() {
          if (searchInput) searchInput.value = '';
          self.searchQuery = '';
          searchClear.style.display = 'none';
          self.applyFilters();
        });
      }

      var btnReset = document.getElementById('wxz-btn-reset');
      if (btnReset) {
        btnReset.addEventListener('click', function() { self.resetFilters(); });
      }
      var btnEmptyReset = document.getElementById('wxz-btn-empty-reset');
      if (btnEmptyReset) {
        btnEmptyReset.addEventListener('click', function() { self.resetFilters(); });
      }

      // 排序按钮
      var sortBtns = root.querySelectorAll('.wxz-sort-btn');
      sortBtns.forEach(function(btn) {
        btn.addEventListener('click', function() {
          sortBtns.forEach(function(b) { b.classList.remove('active'); });
          btn.classList.add('active');
          self.currentSort = btn.getAttribute('data-sort');
          self.applyFilters();
        });
      });

      // 筛选按钮
      root.addEventListener('click', function(e) {
        var btn = e.target.closest('.wxz-filter-btn');
        if (btn) {
          var fType = btn.getAttribute('data-filter');
          var fVal = btn.getAttribute('data-val');
          var group = btn.closest('.wxz-filter-btns');
          if (group) {
            group.querySelectorAll('.wxz-filter-btn').forEach(function(b) { b.classList.remove('active'); });
            btn.classList.add('active');
          }
          self.filters[fType] = fVal;
          self.applyFilters();
        }

        // 复制标准引用
        var copyBtn = e.target.closest('.wxz-btn-copy-cite');
        if (copyBtn) {
          var citeText = copyBtn.getAttribute('data-cite');
          self.copyToClipboard(citeText);
        }

        // 摘要展开/折叠
        var absToggle = e.target.closest('.wxz-abstract-toggle');
        if (absToggle) {
          var pid = absToggle.getAttribute('data-pid');
          self.expandedAbstracts[pid] = !self.expandedAbstracts[pid];
          var bodyEl = document.getElementById('wxz-abs-body-' + pid);
          if (bodyEl) {
            bodyEl.style.display = self.expandedAbstracts[pid] ? 'block' : 'none';
            absToggle.innerHTML = self.expandedAbstracts[pid] ? '收起摘要 ▴' : '展开摘要 ▾';
          }
        }
      });
    },

    resetFilters: function() {
      this.filters = { status: 'all', role: 'all', type: 'all', year: 'all' };
      this.searchQuery = '';
      var searchInput = document.getElementById('wxz-search-input');
      var searchClear = document.getElementById('wxz-search-clear');
      if (searchInput) searchInput.value = '';
      if (searchClear) searchClear.style.display = 'none';

      var root = document.getElementById('wxz-patent-app');
      if (root) {
        root.querySelectorAll('.wxz-filter-btn').forEach(function(b) {
          if (b.getAttribute('data-val') === 'all') {
            b.classList.add('active');
          } else {
            b.classList.remove('active');
          }
        });
      }
      this.applyFilters();
    },

    applyFilters: function() {
      var self = this;
      var q = self.searchQuery;
      var filtered = self.data.filter(function(item) {
        // 状态筛选
        if (self.filters.status === 'granted' && !item.is_granted) return false;
        if (self.filters.status === 'filed' && item.is_granted) return false;

        // 角色筛选
        if (self.filters.role === 'first' && !item.is_first_wang) return false;
        if (self.filters.role === 'co' && item.is_first_wang) return false;

        // 类型筛选
        if (self.filters.type === 'inv' && item.is_utility) return false;
        if (self.filters.type === 'utility' && !item.is_utility) return false;

        // 年份筛选 (支持 2016及以前)
        if (self.filters.year !== 'all') {
          if (self.filters.year === 'lte2016') {
            if (item.app_year > 2016) return false;
          } else {
            if (String(item.app_year) !== String(self.filters.year)) return false;
          }
        }

        // 关键词搜索
        if (q) {
          var matchText = (item.title_zh + ' ' + item.patent_no + ' ' + item.app_no + ' ' + item.inventors_zh_plain + ' ' + item.ipc + ' ' + item.abstract_zh).toLowerCase();
          if (matchText.indexOf(q) === -1) return false;
        }

        return true;
      });

      // 排序
      if (self.currentSort === 'date') {
        filtered.sort(function(a, b) {
          return b.app_date.localeCompare(a.app_date);
        });
      } else if (self.currentSort === 'status') {
        filtered.sort(function(a, b) {
          if (a.is_granted !== b.is_granted) {
            return b.is_granted ? 1 : -1;
          }
          return b.app_date.localeCompare(a.app_date);
        });
      } else {
        var orderMap = {};
        self.data.forEach(function(item, idx) { orderMap[item.app_no] = idx; });
        filtered.sort(function(a, b) {
          return orderMap[a.app_no] - orderMap[b.app_no];
        });
      }

      self.renderCards(filtered);
      self.updateStatsUI(filtered.length);
    },

    updateStatsUI: function(visibleCount) {
      var self = this;
      var totalCount = self.data.length;
      var lblTotal = document.getElementById('wxz-lbl-total-badge');
      var lblFiltered = document.getElementById('wxz-lbl-filtered-badge');

      if (lblTotal) lblTotal.innerText = '全部共 ' + totalCount + ' 项成果';
      if (lblFiltered) {
        if (visibleCount === totalCount) {
          lblFiltered.style.display = 'none';
        } else {
          lblFiltered.style.display = 'inline-block';
          lblFiltered.innerText = '当前显示 ' + visibleCount + ' 项';
        }
      }

      var emptyEl = document.getElementById('wxz-pub-empty');
      var listEl = document.getElementById('wxz-pat-list');
      if (emptyEl && listEl) {
        if (visibleCount === 0) {
          emptyEl.style.display = 'block';
          listEl.style.display = 'none';
        } else {
          emptyEl.style.display = 'none';
          listEl.style.display = 'flex';
        }
      }
    },

    renderCards: function(items) {
      var self = this;
      var listEl = document.getElementById('wxz-pat-list');
      if (!listEl) return;

      var html = '';

      items.forEach(function(item) {
        var cardClass = item.is_granted ? 'is-granted' : 'is-filed';
        var statusBadge = item.is_granted
          ? '<span class="wxz-badge-status badge-granted">★ 已授权</span>'
          : '<span class="wxz-badge-status badge-filed">● 已受理</span>';

        var typeBadge = '<span class="wxz-badge-type">' + item.doc_type_zh + '</span>';
        var isExpanded = self.expandedAbstracts[item.app_no];
        var absDisplay = isExpanded ? 'block' : 'none';
        var absBtnText = isExpanded ? '收起摘要 ▴' : '展开摘要 ▾';

        html += '<div class="wxz-pat-card ' + cardClass + '">';
        // 顶部微标行 + 右侧放置复制引用按钮
        html += '  <div class="wxz-card-top-row">';
        html += '    <div class="wxz-badges-left">';
        html += '      ' + statusBadge;
        html += '      ' + typeBadge;
        html += '    </div>';
        html += '    <button type="button" class="wxz-btn-copy-cite" data-cite="' + item.cite_zh.replace(/"/g, '&quot;') + '" title="点击复制 GB/T 7714 标准引用格式">';
        html += '      <svg viewBox="0 0 24 24" width="13" height="13"><path fill="currentColor" d="M16 1H4c-1.1 0-2 .9-2 2v14h2V3h12V1zm3 4H8c-1.1 0-2 .9-2 2v14c0 1.1.9 2 2 2h11c1.1 0 2-.9 2-2V7c0-1.1-.9-2-2-2zm0 16H8V7h11v14z"/></svg>';
        html += '      <span>复制引用</span>';
        html += '    </button>';
        html += '  </div>';

        // 标题行 (永久固定编号与标题同排呈现)
        html += '  <div class="wxz-card-title-row">';
        html += '    <div class="wxz-card-title">';
        html += '      <span class="wxz-fixed-id">' + item.fixed_id + '.</span>';
        html += '      <span>' + item.title_zh + '</span>';
        html += '    </div>';
        html += '  </div>';

        // 发明人行
        html += '  <div class="wxz-card-authors">' + item.inventors_zh_html + '</div>';

        // 元信息网格 (专利号/申请日/授权日/IPC)
        html += '  <div class="wxz-card-meta-grid">';
        html += '    <div class="wxz-meta-item"><span class="wxz-meta-k">专利/申请号:</span><span class="wxz-meta-v highlight-no">' + item.patent_no + '</span></div>';
        html += '    <div class="wxz-meta-item"><span class="wxz-meta-k">申请日:</span><span class="wxz-meta-v">' + item.app_date + '</span></div>';
        if (item.grant_date) {
          html += '  <div class="wxz-meta-item"><span class="wxz-meta-k">授权公告日:</span><span class="wxz-meta-v">' + item.grant_date + '</span></div>';
        } else if (item.pub_date) {
          html += '  <div class="wxz-meta-item"><span class="wxz-meta-k">公开日:</span><span class="wxz-meta-v">' + item.pub_date + '</span></div>';
        }
        if (item.ipc) {
          html += '  <div class="wxz-meta-item"><span class="wxz-meta-k">IPC:</span><span class="wxz-meta-v">' + item.ipc.split(';')[0] + '</span></div>';
        }
        html += '  </div>';

        // 摘要折叠
        if (item.abstract_zh) {
          html += '  <div class="wxz-abstract-box">';
          html += '    <button type="button" class="wxz-abstract-toggle" data-pid="' + item.app_no + '">' + absBtnText + '</button>';
          html += '    <div class="wxz-abstract-body" id="wxz-abs-body-' + item.app_no + '" style="display:' + absDisplay + ';">' + item.abstract_zh + '</div>';
          html += '  </div>';
        }

        html += '</div>';
      });

      listEl.innerHTML = html;
    },

    copyToClipboard: function(text) {
      var self = this;
      if (navigator.clipboard && window.isSecureContext) {
        navigator.clipboard.writeText(text).then(function() {
          self.showToast('✓ GB/T 7714 标准引用已复制到剪贴板！');
        }).catch(function() {
          self.fallbackCopy(text);
        });
      } else {
        self.fallbackCopy(text);
      }
    },

    fallbackCopy: function(text) {
      var self = this;
      var ta = document.createElement('textarea');
      ta.value = text;
      ta.style.position = 'fixed';
      ta.style.opacity = '0';
      document.body.appendChild(ta);
      ta.focus();
      ta.select();
      try {
        document.execCommand('copy');
        self.showToast('✓ GB/T 7714 标准引用已复制到剪贴板！');
      } catch (err) {
        self.showToast('复制失败，请手动选取复制');
      }
      document.body.removeChild(ta);
    },

    showToast: function(msg) {
      var toast = document.getElementById('wxz-toast');
      if (!toast) return;
      toast.innerText = msg;
      toast.classList.add('show');
      setTimeout(function() {
        toast.classList.remove('show');
      }, 2000);
    }
  };

  window.WxzPatentApp = App;

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', function() { App.init(); });
  } else {
    App.init();
  }
})();
</script>
